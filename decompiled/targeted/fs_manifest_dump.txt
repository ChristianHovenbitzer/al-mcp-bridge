			return SpecializedCollections.EmptyArray<T>();
		}
		ArrayBuilder<T> instance = ArrayBuilder<T>.GetInstance();
		try
		{
			foreach (T val in originalLanguageElements)
			{
				ObjectChangeElement item = new ObjectChangeElement
				{
					Id = val.Id,
					Name = val.Name,
					Namespace = namespaceName,
					Kind = MapeTypeToSymbol(val)
				};
				if (!contained.Contains(item))
				{
					instance.Add(val);
				}
			}
			return instance.ToArray();
		}
		finally
		{
			instance.Free();
		}
	}

	private static SymbolKind MapeTypeToSymbol<T>(T instance)
	{
		if (!(instance is PermissionSetExtensionDefinition))
		{
			if (!(instance is EnumExtensionTypeDefinition))
			{
				if (!(instance is TableExtensionDefinition))
				{
					if (!(instance is NamespaceDefinition))
					{
						if (!(instance is TableDefinition))
						{
							if (!(instance is CodeunitDefinition))
							{
								if (!(instance is PageDefinition))
								{
									if (!(instance is PageExtensionDefinition))
									{
										if (!(instance is PageCustomizationDefinition))
										{
											if (!(instance is ReportDefinition))
											{
												if (!(instance is XmlPortDefinition))
												{
													if (!(instance is QueryDefinition))
													{
														if (!(instance is ProfileDefinition))
														{
															if (!(instance is ProfileExtensionDefinition))
															{
																if (!(instance is ControlAddInDefinition))
																{
																	if (!(instance is EnumTypeDefinition))
																	{
																		if (!(instance is DotNetPackageDefinition))
																		{
																			if (!(instance is InterfaceDefinition))
																			{
																				if (!(instance is PermissionSetDefinition))
																				{
																					if (instance is ReportExtensionDefinition)
																					{
																						return SymbolKind.ReportExtension;
																					}
																					return SymbolKind.Undefined;
																				}
																				return SymbolKind.PermissionSet;
																			}
																			return SymbolKind.Interface;
																		}
																		return SymbolKind.DotNetPackage;
																	}
																	return SymbolKind.Enum;
																}
																return SymbolKind.ControlAddIn;
															}
															return SymbolKind.ProfileExtension;
														}
														return SymbolKind.Profile;
													}
													return SymbolKind.Query;
												}
												return SymbolKind.XmlPort;
											}
											return SymbolKind.Report;
										}
										return SymbolKind.PageCustomization;
									}
									return SymbolKind.PageExtension;
								}
								return SymbolKind.Page;
							}
							return SymbolKind.Codeunit;
						}
						return SymbolKind.Table;
					}
					return SymbolKind.Namespace;
				}
				return SymbolKind.TableExtension;
			}
			return SymbolKind.EnumExtension;
		}
		return SymbolKind.PermissionSetExtension;
	}
