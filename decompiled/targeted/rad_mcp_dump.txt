namespace Microsoft.Dynamics.Nav.EditorServices.Protocol.LanguageServer;

internal class RadLanguageServerCompiler : LanguageServerCompiler
{
	private readonly IObjectChangeManager applicationObjectChangeManager;

	private readonly Predicate<ISymbol> radCompilationFilter;

	internal RadLanguageServerCompiler(IObjectChangeManager applicationObjectChangeManager, Compilation previousCompilation, string projectDirectory, IEnumerable<string> args, IEnumerable<AnalyzerReference> analyzerReferences, string ruleSetPath, string? outFolder, bool externalRulesetsEnabled, IFileSystem fileSystem = null)
		: base(previousCompilation, projectDirectory, args, analyzerReferences, ruleSetPath, outFolder, externalRulesetsEnabled, fileSystem)
	{
		this.applicationObjectChangeManager = applicationObjectChangeManager;
		radCompilationFilter = ShouldMethodCompileApplicationObject;
	}

	public override Compilation GetOrCreateCompilation(CompilerTextWriter consoleOutput, CompilationOptions compilationOptions, ErrorLogger errorLogger)
	{
		return base.PreviousCompilation;
	}

	protected internal override (NavAppPackageWriter Writer, Stream PackageStream) CreateNavAppPackageWriterAndStream(string outputFilePath, CompilerTextWriter consoleOutput)
	{
		if (!base.FileSystem.Exists(outputFilePath))
		{
			if (consoleOutput != null)
			{
				DiagnosticInfo diagnosticInfo = new DiagnosticInfo(base.MessageProvider, 1050, outputFilePath);
				consoleOutput.WriteDiagnostic(diagnosticInfo.Severity, diagnosticInfo.ToString(Culture));
			}
			return (Writer: null, PackageStream: null);
		}
		NavAppManifest navAppManifest = null;
		{
			if (stream == null)
			{
				return (Writer: null, PackageStream: null);
			}
			NavAppPackageReader navAppPackageReader = null;
			try
			{
				navAppPackageReader = NavAppPackageReader.Create(stream, leaveOpen: false, outputFilePath);
				navAppManifest = navAppPackageReader.ReadNavAppManifest();
			}
			catch (InvalidOperationException e)
			{
				LoggingUtilities.WriteMessageFailedException(string.Format(CultureInfo.InvariantCulture, "Rad building failed. We require an existing app, but the one provided in '{0}' did not have a manifest.", outputFilePath), e);
			}
			finally
			{
				navAppPackageReader?.Dispose();
			}
		}
		if (navAppManifest == null)
		{
			if (consoleOutput != null)
			{
				DiagnosticInfo diagnosticInfo2 = new DiagnosticInfo(base.MessageProvider, 565, outputFilePath);
				consoleOutput.WriteDiagnostic(diagnosticInfo2.Severity, diagnosticInfo2.ToString(Culture));
			}
			return (Writer: null, PackageStream: null);
		}
		if (!navAppManifest.AppName.Equals(ProjectManifest.AppManifest.AppName, StringComparison.OrdinalIgnoreCase) || navAppManifest.AppVersion != ProjectManifest.AppManifest.AppVersion || !navAppManifest.AppPublisher.Equals(ProjectManifest.AppManifest.AppPublisher, StringComparison.OrdinalIgnoreCase))
		{
			if (consoleOutput != null)
			{
				DiagnosticInfo diagnosticInfo3 = new DiagnosticInfo(base.MessageProvider, 1051, Path.GetFileName(outputFilePath));
				consoleOutput.WriteDiagnostic(diagnosticInfo3.Severity, diagnosticInfo3.ToString(Culture));
			}
			return (Writer: null, PackageStream: null);
		}
		return base.CreateNavAppPackageWriterAndStream(outputFilePath, consoleOutput);
	}

	protected override PackageModuleOutputterBase CreatePackageModuleOutputter(NavAppPackageWriter packageWriter, Compilation compilation)
	{
		EmitOptions options = base.CreateEmitOptions(compilation.Options);
		return new RadPackageModuleOutputter(applicationObjectChangeManager, ProjectManifest.AppManifest, compilation, options, base.BaseDirectory, packageWriter);
	}

	protected override int RunCoreImpl(CompilerTextWriter consoleOutput, ErrorLogger errorLogger, Compilation compilation, AnalyzerDriver analyzerDriver, IEnumerable<Diagnostic> analyzerExceptionDiagnostics, IEnumerable<AdditionalTextFile> additionalTextFiles, CancellationToken cancellationToken)
	{
		if (base.PreviousCompilation == null)
		{
			return base.RunCoreImpl(consoleOutput, errorLogger, compilation, analyzerDriver, analyzerExceptionDiagnostics, additionalTextFiles, cancellationToken);
		}
		cancellationToken.ThrowIfCancellationRequested();
		ImmutableArray<Diagnostic> immutableArray = base.PreviousCompilation.GetDeclarationDiagnostics(cancellationToken);
		if (!immutableArray.HasAnyErrors())
		{
			immutableArray = ValidateObjectChangesAgainstSymbols();
		}
		if (ReportErrors(immutableArray, consoleOutput, errorLogger, compilation.Options))
		{
			return 1;
		}
		if (ReportErrors(analyzerExceptionDiagnostics, consoleOutput, errorLogger, compilation.Options))
		{
			return 1;
		}
		cancellationToken.ThrowIfCancellationRequested();
		return PackageTheModule(compilation, consoleOutput, cancellationToken, errorLogger, analyzerDriver, null, radCompilationFilter);
	}

	private ImmutableArray<Diagnostic> ValidateObjectChangesAgainstSymbols()
	{
		DiagnosticBag instance = DiagnosticBag.GetInstance();
		foreach (ObjectChangeContext change in applicationObjectChangeManager.Changes)
		{
			Symbol symbol = base.PreviousCompilation.CompiledModule.GetSymbol(change.ObjectChangeElement);
			ErrorCode errorCode = ErrorCode.None;
			switch (change.State)
			{
			case ObjectChangeState.Modified:
			case ObjectChangeState.Added:
				if (symbol == null)
				{
					errorCode = ErrorCode.ERR_ObjectChangeNotFoundInCompilation;
				}
				break;

namespace Microsoft.Dynamics.Nav.EditorServices.Protocol.LanguageServer.Extensions;

internal class ReloadRadFileRequestHandler : RequestHandlerBase<ReloadRadFileRequest>
{
	public override string EndpointName => "al/reloadRadFile";

	internal IFileSystem FileSystem { get; set; }

	internal ReloadRadFileRequestHandler(VsCodeWorkspace workspace, ResponseQueue responseQueue)
		: base(workspace, (IResponseQueue)responseQueue)
	{
	}

	public override Task HandleAsync(ReloadRadFileRequest request, int requestId, CancellationToken cancellationToken)
	{
		ReloadCurrentAndAllProjectReferencesRadFile();
		base.ResponseQueue.Add(ResponseFactory.EmptyResponse(requestId));
		return Task.CompletedTask;
	}

	private void ReloadCurrentAndAllProjectReferencesRadFile()
	{
		Solution currentSolution = base.Workspace.CurrentSolution;
		ProjectId projectId = base.Workspace?.CurrentProjectId;
		base.Workspace?.ObjectChangeDocumentManager.Clear(projectId, keepAdded: false);
		IFileSystem fileSystem = FileSystem ?? Microsoft.Dynamics.Nav.CodeAnalysis.FileSystem.Instance;
		string currentRadPath = base.Workspace.CurrentRadPath;
		if (fileSystem.Exists(currentRadPath))
		{
			base.Workspace.ObjectChangeDocumentManager.ActiveObjectChangeManager.Load(currentRadPath, fileSystem);
		}
		ProjectDependencyGraph projectDependencyGraph = currentSolution.GetProjectDependencyGraph();
		if (projectDependencyGraph == null)
		{
			return;
		}
		foreach (ProjectId item in projectDependencyGraph.GetProjectsThatThisProjectDirectlyDependsOn(projectId))
		{
			IObjectChangeManager objectChangeManager = base.Workspace.ObjectChangeDocumentManager.GetObjectChangeManager(item);
			Project project = currentSolution.GetProject(item);
			if (project != null)
			{
				base.Workspace.ObjectChangeDocumentManager.Clear(project.Id, keepAdded: false);
				string path = project.ProjectFolder.RadPath();
				if (fileSystem.Exists(path))
				{
					objectChangeManager.Load(path, fileSystem);
				}
			}
		}
	}
}
